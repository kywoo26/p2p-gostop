// 실행: node --input-type=module < docs/research/import-boundary-probes.mjs.txt
// 실제 패키지 위치의 일회 파일을 검사하고 finally에서 지운다. #224 도구 설정의 정확한 예외도 구분한다.
import { writeFileSync, unlinkSync } from 'node:fs';
import { spawnSync } from 'node:child_process';

const cases = [];
const add = (dir, source, banned, extension) => cases.push({ dir, source, banned, extension });
for (const target of ['ai', 'protocol', 'web', 'relay-dev', 'sim'])
  add('packages/engine/src', `import '@p2p-gostop/${target}';`, true);
for (const dir of ['packages/ai/src', 'packages/protocol/src', 'tools/sim/src'])
  add(dir, "import '@p2p-gostop/engine';", false);
add('packages/ai/src', "import '@p2p-gostop/protocol';", true);
add('packages/protocol/src', "import '@p2p-gostop/ai';", true);
add('packages/relay-dev/src', "import '@p2p-gostop/protocol';", false);
add('packages/relay-dev/src', "import '@p2p-gostop/engine';", true);
add('packages/relay-dev/test', "import '@p2p-gostop/engine';", false);
add('tools/sim/src', "import '@p2p-gostop/ai';", false);
add('tools/sim/src', "import '@p2p-gostop/protocol';", true);
for (const dir of ['packages/ai/src', 'packages/web/src/lib']) {
  for (const target of ['@p2p-gostop/engine/src/index.ts', '@p2p-gostop/engine/testing/nested', '@p2p-gostop/engine/testing.ts', '../../engine/src/index.ts', '../../../packages/engine/src/index.ts', '../x/../../engine/src/index.ts', '../../engine']) {
    add(dir, `import '${target}';`, true);
    add(dir, `void import('${target}');`, true);
  }
  add(dir, "import type { Seat } from '@p2p-gostop/engine/src/index.ts'; export type { Seat };", true);
  add(dir, "export type { Seat } from '@p2p-gostop/protocol';", dir === 'packages/ai/src');
  add(dir, "export * from '@p2p-gostop/engine/src/index.ts';", true);
  add(dir, "import '@p2p-gostop/engine/testing';", false);
  add(dir, "void import('@p2p-gostop/engine/testing');", false);
  add(dir, "import './index.ts';", false);
}
for (const target of ['engine', 'ai', 'protocol', 'web/net'])
  add('packages/web/src/lib', `import '@p2p-gostop/${target}';`, false);
for (const target of ['relay-dev', 'sim', 'protocol/net', 'ai/testing', 'engine/testing/nested']) {
  add('packages/web/src/lib', `import '@p2p-gostop/${target}';`, true);
  add('packages/web/src/lib', `void import('@p2p-gostop/${target}');`, true);
}
add('packages/web/src/lib', '<script lang="ts">import "@p2p-gostop/engine/src/index.ts";</script>', true);
add('packages/web/src/lib', '<script lang="ts">import "@p2p-gostop/engine";</script>', false);
add('packages/web/e2e', "void import('@p2p-gostop/engine/src/index.ts');", true);
// #98: 백틱 상수와 계산된 경로도 허용하지 않는다. 같은 규칙을 두 AST에서 확인한다.
for (const dir of ['packages/engine/src', 'packages/web/src/lib']) {
  const target = dir.includes('/engine/') ? '@p2p-gostop/protocol' : '@p2p-gostop/relay-dev';
  for (const expression of [
    '`' + target + '`',
    '`' + './index.ts' + '`',
    '`@p2p-gostop/${"protocol"}`',
    '"@p2p-gostop/" + "protocol"',
    'modulePath',
    'String("./index.ts")',
    'true ? "./index.ts" : "./other.ts"',
  ]) add(dir, `const modulePath = './index.ts'; void modulePath; void import(${expression});`, true);
  add(dir, 'void import(("./index.ts"));', false);
}
for (const dir of ['packages/engine/src', 'packages/ai/src', 'packages/protocol/src', 'packages/relay-dev/src', 'packages/relay-dev/test', 'tools/sim/src']) {
  const target = dir === 'tools/sim/src' ? 'protocol' : 'ai';
  for (const extension of ['ts', 'tsx', 'mts', 'cts', 'js', 'jsx', 'mjs', 'cjs']) {
    add(dir, `void import('@p2p-gostop/${target}');`, true, extension);
    add(dir, 'void import("./index.ts");', false, extension);
  }
}
for (const extension of ['js', 'mjs', 'cjs', 'mts', 'cts']) {
  add('packages/web/src/lib', 'void import(`@p2p-gostop/relay-dev`);', true, extension);
  add('packages/web/src/lib', 'void import("@p2p-gostop/engine");', false, extension);
}
add('packages/ai/src', "export * from '@p2p-gostop/protocol';", true, 'mjs');
add('packages/web/src/lib', '<script lang="ts">void import(`@p2p-gostop/relay-dev`);</script>', true);
add('packages/web/src/lib', '<script lang="ts">void import("@p2p-gostop/engine");</script>', false);
add('packages/web/e2e', 'void import(`@p2p-gostop/relay-dev`);', true);
// #224: 같은 도구 경로라도 제품 코드·E2E에서는 금지한다.
for (const dir of ['packages/web/src/lib', 'packages/web/e2e']) {
  add(dir, "import '../../../../tools/lint/lint-imports.mjs';", true);
  add(dir, "void import('../../../../tools/lint/lint-imports.mjs');", true);
}
let passed = 0;
for (const { dir, source, banned, extension } of cases) {
  const web = dir.startsWith('packages/web/');
  const file = `${dir}/__boundary_probe__.${extension ?? (source.startsWith('<script') ? 'svelte' : 'ts')}`;
  writeFileSync(file, source + '\n', { flag: 'wx' });
  try {
    const run = web
      ? spawnSync('../../node_modules/.bin/eslint', [file.slice('packages/web/'.length), '--format', 'json'], { cwd: 'packages/web', encoding: 'utf8' })
      : spawnSync('node_modules/.bin/oxlint', [file, '--format', 'json'], { encoding: 'utf8' });
    if (run.error) throw run.error;
    const report = JSON.parse(run.stdout);
    const diagnostics = web ? report.flatMap((r) => r.messages) : report.diagnostics;
    const found = diagnostics.some((d) => /no-restricted-(imports|syntax)|string-literal-imports/.test(d.ruleId ?? d.code ?? ''));
    if (found !== banned || run.status !== (banned ? 1 : 0))
      throw new Error(JSON.stringify({ file, source, banned, status: run.status, diagnostics, stderr: run.stderr }));
    passed++;
  } finally {
    unlinkSync(file);
  }
}
// 실제 설정 파일을 바꾸지 않고 stdin의 파일명을 설정 경로로 지정한다.
const configCases = [
  ["import '../../tools/lint/lint-imports.mjs';", false],
  ["import '../../tools/lint/other.mjs';", true],
  ["import '../../tools/privacy/privacy-audit.mjs';", true],
  ["import '../../tools/lint/lint-imports.mjs/nested';", true],
  ["import '@p2p-gostop/engine/src/index.ts';", true],
  ["void import('../../tools/lint/lint-imports.mjs');", true],
];
for (const [source, banned] of configCases) {
  const run = spawnSync('../../node_modules/.bin/eslint', ['--stdin', '--stdin-filename', 'eslint.config.js', '--format', 'json'], { cwd: 'packages/web', input: source, encoding: 'utf8' });
  if (run.error) throw run.error;
  const diagnostics = JSON.parse(run.stdout).flatMap((report) => report.messages);
  const found = diagnostics.some((item) => /no-restricted-(imports|syntax)/.test(item.ruleId ?? ''));
  if (found !== banned || run.status !== (banned ? 1 : 0))
    throw new Error(JSON.stringify({ source, banned, status: run.status, diagnostics, stderr: run.stderr }));
  passed++;
}
console.log(`import boundary probes: ${passed}/${cases.length + configCases.length} passed`);
